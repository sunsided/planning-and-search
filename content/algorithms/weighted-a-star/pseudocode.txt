algorithm Weighted-A*(s₀, goal, h, w)
  input: start state s₀, goal test goal, admissible heuristic h, weight w ≥ 1
  output: a path that costs at most w times the optimum, or failure
  g[s₀] ← 0
  OPEN ← {s₀}                                  ▷ ordered by f = g + w · h
  while OPEN ≠ ∅ do
    s ← extract-min(OPEN)
    if goal(s) then
      return path-to(s)
    for each (s′, c) in successors(s) do
      if g[s] + c < g[s′] then                 ▷ g of an unseen state is ∞
        g[s′] ← g[s] + c
        parent[s′] ← s
        insert-or-update(OPEN, s′, g[s′] + w · h(s′))   ▷ w = p/q sorts as q · g + p · h
  return failure
