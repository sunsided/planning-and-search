algorithm FMT*(x₀, X-goal, n, γ)
  input: start state x₀, goal region X-goal, number of samples n, radius constant γ
  output: a path to the goal region through the batch, or failure
  V ← {x₀} ∪ n valid samples                   ▷ plus the goal center when it is valid
  r ← γ · (ln |V| / |V|)^(1/d)                 ▷ d: dimension of the space
  cost[x₀] ← 0
  OPEN ← {x₀}                                  ▷ ordered by cost
  UNVISITED ← V ∖ {x₀}
  while OPEN ≠ ∅ do
    z ← extract-min(OPEN)
    if z ∈ X-goal then
      return path-to(z)
    NEW ← ∅
    for each x in UNVISITED within r of z do
      y ← argmin over open y within r of x of cost[y] + d(y, x)
      if motion-valid(y, x) then               ▷ only the locally best parent is checked
        cost[x] ← cost[y] + d(y, x)
        parent[x] ← y
        NEW ← NEW ∪ {x}
    UNVISITED ← UNVISITED ∖ NEW
    OPEN ← OPEN ∪ NEW                          ▷ z itself is now closed
  return failure
