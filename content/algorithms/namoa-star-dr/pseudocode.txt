algorithm NAMOA*dr(s₀, goal, h)
  input: start state s₀, goal test goal, consistent vector heuristic h
  output: one path per Pareto-optimal cost vector
  G_op[s₀] ← {0}, G_cl[s] ← ∅ for every state s
  OPEN ← {⟨s₀, 0⟩}                       ▷ by f = g + h, lexicographic
  SOLUTIONS ← ∅
  while OPEN ≠ ∅ do
    ⟨s, g⟩ ← extract-min(OPEN)
    if g ∉ G_op[s] then
      continue
    G_op[s] ← G_op[s] ∖ {g}
    if G_cl[s] weakly dominates Tr(g) or Tr(SOLUTIONS) weakly dominates Tr(g + h(s)) then
      continue                           ▷ Tr drops the first objective
    G_cl[s] ← G_cl[s] ∪ {Tr(g)}
    if goal(s) then
      SOLUTIONS ← SOLUTIONS ∪ {path-to(s, g)}
      continue
    for each (s′, c) in successors(s) do
      g′ ← g + c
      if Tr(SOLUTIONS) weakly dominates Tr(g′ + h(s′)) then
        continue
      if G_cl[s′] weakly dominates Tr(g′) then
        continue
      if a vector in G_op[s′] weakly dominates g′ then
        continue                         ▷ open labels: full comparison
      G_op[s′] ← {x ∈ G_op[s′] : g′ does not dominate x} ∪ {g′}
      insert(OPEN, ⟨s′, g′⟩, parent: ⟨s, g⟩)
  return SOLUTIONS
