algorithm A*(s₀, goal, h)
  input: start state s₀, goal test goal, heuristic h
  output: a cheapest path to a goal state, or failure
  g[s₀] ← 0
  OPEN ← {s₀}                                  ▷ ordered by f = g + h
  while OPEN ≠ ∅ do
    s ← extract-min(OPEN)
    if goal(s) then
      return path-to(s)
    for each (s′, c) in successors(s) do
      if g[s] + c < g[s′] then                 ▷ g of an unseen state is ∞
        g[s′] ← g[s] + c
        parent[s′] ← s
        insert-or-update(OPEN, s′, g[s′] + h(s′))   ▷ reopens a closed state
  return failure
