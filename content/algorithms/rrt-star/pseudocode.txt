algorithm RRT*(x₀, n, γ)
  input: start state x₀, number of iterations n, radius constant γ
  output: the cheapest path to the goal region found, or failure
  T ← tree with root x₀
  cost[x₀] ← 0
  for i in 1, …, n do
    x-rand ← sample()
    x-near ← nearest(T, x-rand)
    x-new ← steer(x-near, x-rand)
    if motion-valid(x-near, x-new) then
      r ← min(γ · (ln |T| / |T|)^(1/d), η)      ▷ d: dimension, η: step length
      X-NEAR ← nodes of T within r of x-new
      x-parent ← x-near                          ▷ choose the cheapest parent
      for each x in X-NEAR do
        if cost[x] + c(x, x-new) < cost[x-parent] + c(x-parent, x-new) and motion-valid(x, x-new) then
          x-parent ← x
      add(T, x-new, parent: x-parent)
      cost[x-new] ← cost[x-parent] + c(x-parent, x-new)   ▷ c: the objective's motion cost, d by default
      for each x in X-NEAR do                    ▷ rewire through x-new
        if cost[x-new] + c(x-new, x) < cost[x] and motion-valid(x-new, x) then
          reparent(T, x, x-new)                  ▷ updates the costs below x
  return path-to(cheapest goal node of T), or failure if there is none
function Grow-Forest(x₀, n, γ, RNGS)
  ▷ feature parallel: one tree per random stream, merged into the first
  in parallel do
    for each ρ in RNGS do
      T[ρ] ← RRT*(x₀, n, γ) with random stream ρ
  T ← T[first of RNGS]
  for each T′ in the other trees, in the order of RNGS do
    for each x in the nodes of T′, parents before children do
      X-NEAR ← nodes of T within r of x, and x's parent from T′   ▷ that edge T′ already checked
      add x under the cheapest valid candidate of X-NEAR, then rewire through x
  return T
