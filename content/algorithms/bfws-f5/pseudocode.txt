algorithm BFWS-f5(s₀, k)
  input: initial state s₀, width bound k
  output: a plan to a goal state, or failure
  if goal(s₀) then
    return empty plan
  RELEVANT[s₀] ← relaxed-plan-features(s₀) ∖ features(s₀)
  REACHED[s₀] ← ∅
  SEEN ← {s₀}
  OPEN ← {s₀}                        ▷ ordered by ⟨w, g⟩, then by generation order
  w[s₀] ← novelty(s₀)
  while OPEN ≠ ∅ do
    s ← extract-min(OPEN)
    for each (a, s′) in successors(s) do
      if s′ ∈ SEEN then
        continue
      SEEN ← SEEN ∪ {s′}
      parent[s′] ← (s, a)
      if unsatisfied-goals(s′) < unsatisfied-goals(s) then
        RELEVANT[s′] ← relaxed-plan-features(s′) ∖ features(s′)   ▷ a new relaxed plan
        REACHED[s′] ← ∅
      else
        RELEVANT[s′] ← RELEVANT[s]
        REACHED[s′] ← REACHED[s] ∪ (features(s′) ∩ RELEVANT[s])
      w[s′] ← novelty(s′)
      if goal(s′) then
        return plan-to(s′)
      insert(OPEN, s′, ⟨w[s′], unsatisfied-goals(s′)⟩)
  return failure
function novelty(s)
  g ← unsatisfied-goals(s)
  r ← |REACHED[s]|
  w ← evaluate-and-insert(TABLE[g, r], features(s))   ▷ one table per pair (g, r)
  if w = nil then
    w ← k + 1
  return w
