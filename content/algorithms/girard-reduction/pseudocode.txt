algorithm Girard-Reduction(Z, ρ)
  input: zonotope Z = ⟨c, G⟩ with p generators in n dimensions, order ρ ≥ 1
  output: a zonotope that encloses Z and has at most ⌊ρ · n⌋ generators
  if p ≤ ρ · n then
    return Z
  for each generator g in G do
    m[g] ← ‖g‖₁ − ‖g‖∞                    ▷ zero for a generator along an axis
  sort-descending(G, m)
  k ← ⌊ρ · n⌋ − n
  K ← the first k generators of G        ▷ kept unchanged
  D ← the other p − k generators         ▷ dropped
  for i in 1, …, n do
    r_i ← ∑ over g ∈ D of |g_i|           ▷ box radius along axis i
  B ← diag(r₁, …, r_n)
  return ⟨c, K ∪ columns(B)⟩
