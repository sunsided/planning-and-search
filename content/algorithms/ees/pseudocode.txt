algorithm EES(S₀, goal, h, ĥ, d̂, w)
  input: start states S₀, goal test goal, admissible heuristic h, cost estimate ĥ, distance estimate d̂, weight w ≥ 1
  output: a path to a goal that costs at most w times the optimum, or failure
  for each s in S₀ do
    g[s] ← 0
    insert(OPEN, s)
  while OPEN ≠ ∅ do
    s ← select-node(OPEN)
    remove(OPEN, s)
    if goal(s) then
      return path-to(s)
    for each (s′, c) in successors(s) do
      if g[s] + c < g[s′] then
        g[s′] ← g[s] + c
        parent[s′] ← s
        insert-or-update(OPEN, s′)             ▷ reopens a closed state
  return failure
function select-node(OPEN)
  best-f ← the state of OPEN with the smallest f = g + h      ▷ the cleanup order
  best-f̂ ← the state of OPEN with the smallest f̂ = g + ĥ
  FOCAL ← { s ∈ OPEN : f̂(s) ≤ w · f̂(best-f̂) }
  best-d̂ ← the state of FOCAL with the smallest d̂
  B ← w · f(best-f)
  if f̂(best-d̂) ≤ B then
    return best-d̂
  else if f̂(best-f̂) ≤ B then
    return best-f̂
  return best-f
